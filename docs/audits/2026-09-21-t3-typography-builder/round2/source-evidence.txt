17:   frame?: Pick<PageFrameModel, 'templateId' | 'pageSize'>,
18: ): 'paper' | 'web' {
19:   if (frame?.templateId === 'screen_note') return 'web';
20:   if (frame?.templateId === 'a4_portrait' || frame?.templateId === 'letter_portrait') return 'paper';
21:   return frame?.pageSize === 'Custom' ? 'web' : 'paper';

24: export function createPageFrameDefaultTypographyProfile(
25:   frame?: Partial<PageFrameModel>,
26: ): DocumentTypographyProfile {
27:   const family = getPageFrameTypographyFamily(frame);
28:   const baseline = DEFAULT_DOCUMENT_TYPOGRAPHY_PROFILE;
29:   const pageSize = frame?.pageSize
30:     ?? (frame?.templateId === 'letter_portrait' ? 'Letter' : 'A4');
31:   const printProfile = createPageFramePrintProfile(pageSize);
32:   const { physicalScale } = getPageFramePhysicalMapping(
33:     pageSize,
34:     frame?.width ?? printProfile.width,
35:     frame?.templateId,
36:     frame?.paperSizeReferenceWidth,
37:   );
38:   // Paper defaults use a quarter-point type scale and a more open reading rhythm.
39:   // Physical mapping and the single normalization pass stay shared with print.
40:   const fontSizePx = family === 'paper' ? (10.75 * 96 / 72) / physicalScale : 16;
41:   const ratio = fontSizePx / baseline.fontSizePx;
42:   return normalizeDocumentTypographyProfile({
43:     ...baseline,
44:     profileId: family === 'paper' ? 'paper-document' : 'web-document',
45:     fontSizePx,
46:     lineHeightPx: family === 'paper' ? fontSizePx * 1.65 : baseline.lineHeightPx * ratio,
47:     paragraphSpacingPx: baseline.paragraphSpacingPx * ratio,
48:     averageCharWidthPx: baseline.averageCharWidthPx * ratio,
49:   });
50: }
51: 
52: export function resolveEffectiveDocumentTypographyProfile({
53:   metadata,
54:   pageFrames,
55:   hydratedProfile,
56: }: {
57:   surfaceMode: NoteCanvasMode;
58:   metadata: Record<string, unknown> | null | undefined;
59:   pageFrames: PageFrameModel[] | undefined;
60:   hydratedProfile: DocumentTypographyProfile;
61: }): DocumentTypographyProfile {
62:   if (hasDocumentTypographyProfileOverride(metadata)) {
63:     return hydratedProfile;
64:   }
65:   // A note has one effective profile: selection changes never change the family.
66:   return createPageFrameDefaultTypographyProfile(pageFrames?.[0]);
